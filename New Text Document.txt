        <span>Stamina</span>
        <div class="bar"><div id="staminaFill" class="bar-fill green"></div></div>
      </div>
      <div id="objective">Objective: Clear the Camps</div>
    </div>

    <div id="status" class="panel">
      <div><strong>Enemies:</strong> <span id="enemyCount">0</span></div>

  <canvas id="game" width="1280" height="720"></canvas>

  <script>
    const canvas = document.getElementById("game");
    const ctx = canvas.getContext("2d");

    const healthFill = document.getElementById("healthFill");
    const staminaFill = document.getElementById("staminaFill");
    const enemyCount = document.getElementById("enemyCount");
    const objectiveText = document.getElementById("objective");

    const world = { w: 2200, h: 2200 };
    const camera = { x: 0, y: 0 };

    const keys = {};
    let pointerDown = false;
    let mouse = { x: canvas.width / 2, y: canvas.height / 2 };

    const player = {
      x: world.w / 2,
      y: world.h / 2,
      radius: 18,
      speed: 280,
      sprintSpeed: 430,
      health: 100,
      maxHealth: 100,
      stamina: 100,
      maxStamina: 100,
      fireCooldown: 0,
      damage: 22,
      bulletSpeed: 900,
      angle: 0
    };

    let enemies = [];
    let bullets = [];
    let trees = [];
    let camps = [];
    let gameOver = false;
    let win = false;
    let fireShake = 0;

    function clamp(value, min, max) {
      return Math.min(Math.max(value, min), max);
    }

    function rand(min, max) {
      return Math.random() * (max - min) + min;
    }

    function distance(a, b) {
      return Math.hypot(a.x - b.x, a.y - b.y);
    }

    function generateTrees() {
      trees = [];
      for (let i = 0; i < 180; i++) {
        const x = rand(40, world.w - 40);
        const y = rand(40, world.h - 40);
        const size = rand(18, 42);
        trees.push({ x, y, size, sway: rand(0, Math.PI * 2) });
      }
    }

    function generateCamps() {
      camps = [];
      const campPositions = [
        { x: 320, y: 340 }, { x: 620, y: 820 }, { x: 1060, y: 360 },
        { x: 1500, y: 870 }, { x: 1750, y: 1420 }, { x: 420, y: 1600 }
      ];

      campPositions.forEach((center, index) => {
        const camp = {
          x: center.x,
          y: center.y,
          radius: 95,
          color: index % 2 === 0 ? "#e78b42" : "#d4483d",
          enemies: []
        };

        for (let i = 0; i < 4; i++) {
          const angle = (Math.PI * 2 * i) / 4 + rand(-0.5, 0.5);
          const enemy = {
            x: center.x + Math.cos(angle) * rand(35, 80),
            y: center.y + Math.sin(angle) * rand(35, 80),
            radius: 15,
            maxHealth: 40,
            health: 40,
            speed: 80,
            attackCooldown: 0,
            wanderTimer: rand(0.6, 2),
            wanderAngle: rand(0, Math.PI * 2),
            color: "#db5a4a",
            active: false
          };
          camp.enemies.push(enemy);
        }

        camps.push(camp);
      });

      enemies = [];
      camps.forEach(camp => enemies.push(...camp.enemies));
    }

    function updateHud() {
      const healthPct = (player.health / player.maxHealth) * 100;
      const staminaPct = (player.stamina / player.maxStamina) * 100;

      healthFill.style.width = healthPct + "%";
      staminaFill.style.width = staminaPct + "%";
      enemyCount.textContent = enemies.length;

      if (gameOver) {
        objectiveText.textContent = "Mission Failed";
      } else if (win) {
        objectiveText.textContent = "Island Secured";
      } else {
        objectiveText.textContent = "Objective: Clear the Camps";
      }
    }

    function shoot() {
      if (gameOver || win) return;
      if (player.fireCooldown > 0) return;

      const angle = player.angle;
      bullets.push({
        x: player.x + Math.cos(angle) * 28,
        y: player.y + Math.sin(angle) * 28,
        vx: Math.cos(angle) * player.bulletSpeed,
        vy: Math.sin(angle) * player.bulletSpeed,
        radius: 4,
        life: 1.0,
        damage: player.damage
      });

      player.fireCooldown = 0.12;
      fireShake = 0.22;
    }

    function handleInput(dt) {
      if (gameOver || win) return;

      let moveX = 0;
      let moveY = 0;

      if (keys["w"] || keys["arrowup"]) moveY -= 1;
      if (keys["s"] || keys["arrowdown"]) moveY += 1;
      if (keys["a"] || keys["arrowleft"]) moveX -= 1;
      if (keys["d"] || keys["arrowright"]) moveX += 1;

      const moving = moveX !== 0 || moveY !== 0;
      const sprinting = keys["shift"] && moving && player.stamina > 0;

      if (moving) {
        const len = Math.hypot(moveX, moveY) || 1;
        moveX /= len;
        moveY /= len;

        const speed = sprinting ? player.sprintSpeed : player.speed;
        player.x += moveX * speed * dt;
        player.y += moveY * speed * dt;

        if (sprinting) {
          player.stamina = Math.max(0, player.stamina - 34 * dt);
        } else {
          player.stamina = Math.min(player.maxStamina, player.stamina + 16 * dt);
        }
      } else {
        player.stamina = Math.min(player.maxStamina, player.stamina + 22 * dt);
      }

      player.x = clamp(player.x, player.radius, world.w - player.radius);
      player.y = clamp(player.y, player.radius, world.h - player.radius);

      const worldMouseX = mouse.x + camera.x;
      const worldMouseY = mouse.y + camera.y;
      player.angle = Math.atan2(worldMouseY - player.y, worldMouseX - player.x);
    }

    function updateBullets(dt) {
      for (let i = bullets.length - 1; i >= 0; i--) {
        const bullet = bullets[i];
        bullet.x += bullet.vx * dt;
        bullet.y += bullet.vy * dt;
        bullet.life -= dt;

        if (
          bullet.x < -20 || bullet.x > world.w + 20 ||
          bullet.y < -20 || bullet.y > world.h + 20 ||
          bullet.life <= 0
        ) {
          bullets.splice(i, 1);
          continue;
        }

        for (let j = enemies.length - 1; j >= 0; j--) {
          const enemy = enemies[j];
          if (distance(bullet, enemy) < bullet.radius + enemy.radius) {
            enemy.health -= bullet.damage;
            bullets.splice(i, 1);

            if (enemy.health <= 0) {
              enemies.splice(j, 1);
            }
            break;
          }
        }
      }
    }

    function updateEnemies(dt) {
      for (let i = 0; i < enemies.length; i++) {
        const enemy = enemies[i];
        enemy.attackCooldown = Math.max(0, enemy.attackCooldown - dt);

        const dx = player.x - enemy.x;
        const dy = player.y - enemy.y;
        const dist = Math.hypot(dx, dy);

        if (dist < 360) {
          enemy.active = true;
          const dirX = dx / (dist || 1);
          const dirY = dy / (dist || 1);

          if (dist > enemy.radius + player.radius + 10) {
            enemy.x += dirX * enemy.speed * dt;
            enemy.y += dirY * enemy.speed * dt;
          }

          if (dist < enemy.radius + player.radius + 12 && enemy.attackCooldown <= 0) {
            player.health -= 10;
            enemy.attackCooldown = 1.1;
          }
        } else {
          enemy.active = false;
          enemy.wanderTimer -= dt;
          if (enemy.wanderTimer <= 0) {
            enemy.wanderTimer = rand(0.8, 2.2);
            enemy.wanderAngle = rand(0, Math.PI * 2);
          }

          enemy.x += Math.cos(enemy.wanderAngle) * enemy.speed * 0.25 * dt;
          enemy.y += Math.sin(enemy.wanderAngle) * enemy.speed * 0.25 * dt;
        }

        enemy.x = clamp(enemy.x, 20, world.w - 20);
        enemy.y = clamp(enemy.y, 20, world.h - 20);
      }

      if (player.health <= 0) {
        gameOver = true;
      }

      if (enemies.length === 0 && !win && !gameOver) {
        win = true;
      }
    }

    function updateCamera() {
      camera.x = clamp(player.x - canvas.width / 2, 0, world.w - canvas.width);
      camera.y = clamp(player.y - canvas.height / 2, 0, world.h - canvas.height);
    }

    function drawGround() {
      ctx.fillStyle = "#1d5d40";
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      const startX = camera.x;
      const startY = camera.y;
      ctx.save();
      ctx.translate(-camera.x, -camera.y);

      ctx.fillStyle = "#2c7a50";
      ctx.fillRect(0, 0, world.w, world.h);

      // Paths and noise
      for (let y = 0; y < world.h; y += 35) {
        for (let x = 0; x < world.w; x += 35) {
          ctx.fillStyle = (x + y) % 70 === 0 ? "rgba(255,255,255,0.03)" : "rgba(0,0,0,0.02)";
          ctx.fillRect(x, y, 20, 20);
        }
      }

      // Water / coast ring
      ctx.strokeStyle = "rgba(46, 110, 115, 0.66)";
      ctx.lineWidth = 18;
      ctx.beginPath();
      ctx.arc(200, 200, 150, 0, Math.PI * 2);
      ctx.stroke();

      // Camps
      camps.forEach(camp => {
        ctx.beginPath();
        ctx.fillStyle = "rgba(255, 198, 100, 0.18)";
        ctx.arc(camp.x, camp.y, camp.radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.strokeStyle = "rgba(255, 180, 60, 0.7)";
        ctx.lineWidth = 2;
        ctx.stroke();
      });

      // Trees
      trees.forEach(tree => {
        const x = tree.x;
        const y = tree.y;
        const s = tree.size;

        ctx.fillStyle = "#183a2d";
        ctx.fillRect(x - 4, y, 8, s * 0.9);

        ctx.beginPath();
        ctx.fillStyle = "#224f38";
        ctx.arc(x, y, s * 0.55, 0, Math.PI * 2);
        ctx.fill();

        ctx.beginPath();
        ctx.fillStyle = "#295c3f";
        ctx.arc(x - s * 0.25, y - s * 0.12, s * 0.38, 0, Math.PI * 2);
        ctx.arc(x + s * 0.2, y - s * 0.08, s * 0.42, 0, Math.PI * 2);
        ctx.arc(x, y - s * 0.26, s * 0.44, 0, Math.PI * 2);
        ctx.fill();
      });

      ctx.restore();
    }

    function drawPlayer() {
      const sx = player.x - camera.x;
      const sy = player.y - camera.y;

      ctx.save();
      ctx.translate(sx, sy);

      ctx.beginPath();
      ctx.fillStyle = "#3b8ee5";
      ctx.arc(0, 0, player.radius, 0, Math.PI * 2);
      ctx.fill();

      // Gun
      ctx.strokeStyle = "#d6ede7";
      ctx.lineWidth = 5;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(Math.cos(player.angle) * 22, Math.sin(player.angle) * 22);
      ctx.stroke();

      // Aim line
      ctx.strokeStyle = "rgba(255,255,255,0.35)";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(Math.cos(player.angle) * 50, Math.sin(player.angle) * 50);
      ctx.stroke();

      ctx.restore();
    }

    function drawEnemies() {
      enemies.forEach(enemy => {
        const sx = enemy.x - camera.x;
        const sy = enemy.y - camera.y;

        ctx.beginPath();
        ctx.fillStyle = enemy.active ? "#ff6a4d" : "#c9483d";
        ctx.arc(sx, sy, enemy.radius, 0, Math.PI * 2);
        ctx.fill();

        const healthRatio = enemy.health / enemy.maxHealth;
        ctx.fillStyle = "rgba(0,0,0,0.45)";
        ctx.fillRect(sx - 18, sy - 26, 36, 6);
        ctx.fillStyle = "#6ae19a";
        ctx.fillRect(sx - 18, sy - 26, 36 * healthRatio, 6);
      });
    }

    function drawBullets() {
      bullets.forEach(bullet => {
        const sx = bullet.x - camera.x;
        const sy = bullet.y - camera.y;
        ctx.beginPath();
        ctx.fillStyle = "#f5d76e";
        ctx.arc(sx, sy, bullet.radius, 0, Math.PI * 2);
        ctx.fill();
      });
    }

    function drawVignette() {
      ctx.fillStyle = "rgba(0,0,0,0.2)";
      ctx.fillRect(0, 0, canvas.width, canvas.height);
    }

    function drawCenterReticle() {
      const cx = canvas.width / 2;
      const cy = canvas.height / 2;
      ctx.strokeStyle = "rgba(255,255,255,0.8)";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(cx - 9, cy);
      ctx.lineTo(cx + 9, cy);
      ctx.moveTo(cx, cy - 9);
      ctx.lineTo(cx, cy + 9);
      ctx.stroke();
    }

    function drawGameOver() {
      if (!gameOver && !win) return;
      ctx.fillStyle = "rgba(0,0,0,0.5)";
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      ctx.fillStyle = "#f5f5f5";
      ctx.textAlign = "center";
      ctx.font = "bold 52px Arial";
      ctx.fillText(gameOver ? "Mission Failed" : "Island Secured", canvas.width / 2, canvas.height / 2 - 20);

      ctx.font = "22px Arial";
      ctx.fillStyle = "#dfe9e5";
      ctx.fillText("Press R to restart", canvas.width / 2, canvas.height / 2 + 28);
    }

    function loop() {
      const now = performance.now();
      const dt = Math.min(0.033, (now - (loop.lastTime || now)) / 1000);
      loop.lastTime = now;

      if (!gameOver && !win) {
        handleInput(dt);
        updateBullets(dt);
        updateEnemies(dt);
        player.fireCooldown = Math.max(0, player.fireCooldown - dt);
        if (fireShake > 0) fireShake = Math.max(0, fireShake - dt);
      }

      updateCamera();
      updateHud();

      drawGround();
      drawBullets();
      drawEnemies();
      drawPlayer();
      drawCenterReticle();
      drawGameOver();

      requestAnimationFrame(loop);
    }

    window.addEventListener("keydown", (e) => {
      const key = e.key.toLowerCase();
      keys[key] = true;

      if (key === " " || key === "e") {
        shoot();
      }

      if (key === "r" && (gameOver || win)) {
        resetGame();
      }
    });

    window.addEventListener("keyup", (e) => {
      keys[e.key.toLowerCase()] = false;
    });

    canvas.addEventListener("mousemove", (e) => {
      const rect = canvas.getBoundingClientRect();
      mouse.x = (e.clientX - rect.left) * (canvas.width / rect.width);
      mouse.y = (e.clientY - rect.top) * (canvas.height / rect.height);
    });

    canvas.addEventListener("mousedown", () => {
      pointerDown = true;
      shoot();
    });

    window.addEventListener("mouseup", () => {
      pointerDown = false;
    });

    function resetGame() {
      gameOver = false;
      win = false;
      player.x = world.w / 2;
      player.y = world.h / 2;
      player.health = player.maxHealth;
      player.stamina = player.maxStamina;
      player.fireCooldown = 0;
      generateTrees();
      generateCamps();
      updateHud();
    }

    generateTrees();
    generateCamps();
    updateHud();
    requestAnimationFrame(loop);
  </script>
